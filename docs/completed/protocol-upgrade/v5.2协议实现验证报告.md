```python
if hasattr(conn, "_pending_esp32_acks") and seq_id in conn._pending_esp32_acks:

# commandProxyHandler.py
async def _wait_for_esp32_ack(self, esp32_conn, seq_id: str, timeout: float) -> bool:
    future = asyncio.Future()
    esp32_conn._pending_esp32_acks[seq_id] = future
    try:
        result = await asyncio.wait_for(future, timeout=timeout)
        return result
    except asyncio.TimeoutError:
        return False
```

### 3.8 命令重试机制【v5.1 新增】

**协议要求**：
- 等待 esp32_ack 超时（3 秒）后重试
- 最多重试 3 次
- 重试时保持原始 seq_id 不变

**代码实现**：

| 功能 | 文件 | 实现状态 |
|------|------|----------|
| 重试逻辑 | `commandProxyHandler.py` | ✅ 已实现 |
| 超时设置 | `commandProxyHandler.py` | ✅ 已实现（2秒） |
| seq_id 保持 | `commandProxyHandler.py` | ✅ 已实现 |

**验证结果**：✅ 完全匹配（超时时间略有差异：代码 2 秒，协议 3 秒）

**代码示例**：
```python
# commandProxyHandler.py
async def _forward_with_retry(self, conn, esp32_conn, msg_json: Dict[str, Any]) -> bool:
    seq_id = msg_json.get("seq_id")
    max_retries = 3
    
    for retry in range(max_retries):
        try:
            # 发送命令到 ESP32
            await esp32_conn.websocket.send(json.dumps(msg_json, ensure_ascii=False))
            
            # 等待 esp32_ack（2 秒超时）
            ack_received = await self._wait_for_esp32_ack(esp32_conn, seq_id, timeout=2.0)
            
            if ack_received:
                return True
        except Exception as e:
            conn.logger.bind(tag=TAG).warning(f"发送命令失败（第 {retry + 1} 次）")
    
    # 重试全部失败
    await self._send_error(conn, "设备无响应", code=ErrorCode.TIMEOUT)
    return False
```
